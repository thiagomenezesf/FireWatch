// Declaração das variáveis que faltavam no código original
float latitude = 0.0000;
float longitude = 0.0000;
String regiao = "";
String bioma = "";

void setup() { 
  Serial.begin(9600);

  while (!Serial) {
    ; // Espera a conexão ser estabelecida
  }

  delay(1000); 
  
  // Inicializa o gerador de números aleatórios com uma leitura analógica flutuante
  randomSeed(analogRead(0)); 
} 

void loop() {
  int arduinoValue = random(1, 31);
  String arduino = "Arduino_0" + String(arduinoValue);

  switch(arduinoValue) {
    case 1 ... 5:
      regiao = "Mirante Serra da Paulista";
      bioma = "Mata Atlantica";
      break;
    case 6 ... 10:
      regiao = "Região Pedra Balão";
      bioma = "Cerrado";
      break;
    case 11 ... 15:
      regiao = "Pesqueiro Bambu Amarelo";
      bioma = "Mata Atlantica";
      break;
    case 16 ... 20:
      regiao = "Capelinha Nossa Senhora";
      bioma = "Mata Atlantica";
      break;
    case 21 ... 25:
      regiao = "Vinicola Lancellotti";
      bioma = "Mata Atlantica";
      break;
    case 26 ... 30:
      regiao = "Cruz Cruzeiro do Sul";
      bioma = "Cerrado";
      break;
    default:
      regiao = "Regiao Desconhecida";
      bioma = "Ignorado";
      break;
  }

  latitude = listaLatitudes[arduinoValue - 1];
  longitude = listaLongitudes[arduinoValue - 1]; 

  // random(min, max) gera números inteiros. 
  // Dividimos por 10.0 para obter números decimais (float).
  float temperatura = random(150, 400) / 10.0; // Gera temperatura entre 15.0 e 40.0 °C
  float umidade = random(300, 900) / 10.0;     // Gera umidade entre 30.0% e 90.0%

  if (isnan(temperatura) || isnan(umidade)) { 
    Serial.println("ERRO"); 
  } else { 
    // Corrigido para Serial.print para manter tudo na mesma linha (conforme o padrão CSV)
    Serial.print(arduino); 
    Serial.print(","); 
    Serial.print(temperatura); 
    Serial.print(","); 
    Serial.print(umidade); 
    Serial.print(","); 
    Serial.print(latitude, 6); 
    Serial.print(","); 
    Serial.print(longitude, 6); // println no final para pular a linha para o próximo envio
    Serial.print(",");
    Serial.print(regiao);
    Serial.print(",");
    Serial.println(bioma);
  } 
  
  delay(7000); 
}
