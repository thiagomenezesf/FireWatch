#include <DHT.h>

#define DHTPIN A1
#define DHTTYPE DHT22

const char* arduinoId = arduino-01;

const float latitude = -21.123123213;
const float longitude = -46.89798798;

DHT dht(DHTPIN, DHTTYPE);

void setup() {
  Serial.begin(9600);
  dht.begin();

  delay(2000);
}

void loop() {
  float temperatura = dht.readTemperature();
  float umidade = dht.readHumidity();

  if (isnan(temperatura) || isnan(umidade)) {
    Serial.println("ERRO");
  } else {
    Serial.println(arduino);
    Serial.println(",");
    Serial.print(temperatura);
    Serial.print(",");
    Serial.print(umidade);
    Serial.print(",");
    Serial.print(latitude, 6);
    Serial.print(",");
    Serial.println(longitude, 6);
  }

  delay(10000);
}
